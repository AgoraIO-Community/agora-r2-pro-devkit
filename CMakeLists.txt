cmake_minimum_required(VERSION 3.25)
project(mybot_rv1126b VERSION 0.2.2 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(RV1126B_HOST_ONLY "Build host tests and UI preview without target SDK" OFF)
option(RV1126B_BUILD_TESTS "Build host tests" ${RV1126B_HOST_ONLY})
option(RV1126B_ENABLE_UBSAN "Instrument host tests with UBSan" OFF)
option(MYBOT_ENABLE_VIDEO "Include V4L2/MPP JPEG uplink" OFF)
set(RV1126B_LANGUAGE "zh-CN" CACHE STRING "UI language: zh-CN or en-US")
set_property(CACHE RV1126B_LANGUAGE PROPERTY STRINGS zh-CN en-US)
if(RV1126B_LANGUAGE STREQUAL "zh-CN")
    set(RV1126B_CHINESE 1)
elseif(RV1126B_LANGUAGE STREQUAL "en-US")
    set(RV1126B_CHINESE 0)
else()
    message(FATAL_ERROR "RV1126B_LANGUAGE must be zh-CN or en-US")
endif()
if(RV1126B_ENABLE_UBSAN)
    if(NOT RV1126B_HOST_ONLY)
        message(FATAL_ERROR "UBSan is supported only for host builds")
    endif()
    add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    add_link_options(-fsanitize=undefined -fno-sanitize-recover=all)
endif()

find_package(Threads REQUIRED)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
include(src/platform/boards/rv1126b-test2-rk805b-v10/board.cmake)
include(cmake/dependencies/public-headers.cmake)
include(cmake/dependencies/lvgl.cmake)
add_subdirectory(src/ui)
if(NOT RV1126B_HOST_ONLY)
    include(cmake/dependencies/sdk.cmake)
endif()
add_subdirectory(src/platform)
if(RV1126B_HOST_ONLY)
    add_subdirectory(tools/ui_preview)
    if(RV1126B_BUILD_TESTS)
        enable_testing()
        add_subdirectory(tests)
    endif()
else()
    add_subdirectory(src/application)
endif()
