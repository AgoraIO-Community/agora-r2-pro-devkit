set(platform_root "${PROJECT_SOURCE_DIR}/src/platform")
set(display_dir "${platform_root}/src/drivers/display")
set(input_dir "${platform_root}/src/drivers/input")
set(support_dir "${CMAKE_CURRENT_SOURCE_DIR}/support")
find_package(OpenSSL REQUIRED)

add_executable(test-pixels unit/test_pixels.c)
target_include_directories(test-pixels PRIVATE "${display_dir}")
target_link_libraries(test-pixels PRIVATE mybot_pixels)
add_test(NAME pixels COMMAND test-pixels)
add_executable(test-input unit/test_input.c "${input_dir}/input_logic.c")
target_include_directories(test-input PRIVATE support/mocks "${input_dir}")
target_link_libraries(test-input PRIVATE mybot::public_headers)
add_test(NAME input-semantics COMMAND test-input)
add_executable(test-input-worker integration/test_input_worker.c "${input_dir}/input_logic.c")
target_include_directories(test-input-worker PRIVATE support/mocks "${input_dir}"
    "${platform_root}/include" "${platform_root}/src/internal")
target_link_libraries(test-input-worker PRIVATE mybot::public_headers Threads::Threads)
add_test(NAME input-worker COMMAND test-input-worker)
add_executable(test-renderer unit/test_renderer.c)
target_include_directories(test-renderer PRIVATE "${display_dir}")
target_link_libraries(test-renderer PRIVATE mybot_ui mybot_lvgl mybot_pixels)
add_test(NAME renderer COMMAND test-renderer)
add_executable(test-identity unit/test_identity.c ../src/application/device_identity.c)
target_include_directories(test-identity PRIVATE ../src/application)
target_link_libraries(test-identity PRIVATE OpenSSL::Crypto)
add_test(NAME identity COMMAND test-identity)
add_executable(test-ui-runtime integration/test_ui_runtime.c "${platform_root}/src/services/display_runtime.c")
target_include_directories(test-ui-runtime PRIVATE "${platform_root}/include" "${display_dir}")
target_link_libraries(test-ui-runtime PRIVATE mybot_ui_contract Threads::Threads)
add_test(NAME ui-runtime COMMAND test-ui-runtime)

# The standard host suite is complete or fails configuration. No silent skips.
set(RV1126B_TEST_DRM_INCLUDE "" CACHE PATH "Explicit libdrm headers for host tests")
find_path(RV1126B_TEST_XF86DRM xf86drm.h HINTS "${RV1126B_TEST_DRM_INCLUDE}" PATH_SUFFIXES libdrm REQUIRED)
find_path(RV1126B_TEST_DRM_UAPI drm.h HINTS "${RV1126B_TEST_DRM_INCLUDE}"
    "${RV1126B_TEST_XF86DRM}" PATH_SUFFIXES libdrm REQUIRED)
add_executable(test-drm integration/test_drm.c)
target_include_directories(test-drm PRIVATE "${display_dir}" "${RV1126B_TEST_XF86DRM}" "${RV1126B_TEST_DRM_UAPI}")
target_link_libraries(test-drm PRIVATE mybot_pixels)
add_test(NAME drm-lifecycle COMMAND test-drm)

add_executable(test-config unit/test_config.c ../src/application/config.c)
target_include_directories(test-config PRIVATE ../src/application support/mocks "${platform_root}/include" "${RV1126B_BOARD_INCLUDE}")
target_compile_definitions(test-config PRIVATE RV1126B_VERSION="${PROJECT_VERSION}" RV1126B_LOCALE="${RV1126B_LANGUAGE}" MYBOT_ENABLE_VIDEO=0)
target_link_libraries(test-config PRIVATE mybot::public_headers)
add_test(NAME app-config COMMAND test-config)
add_executable(test-runtime-guard integration/test_runtime_guard.c ../src/application/runtime_guard.c)
target_include_directories(test-runtime-guard PRIVATE ../src/application)
add_test(NAME runtime-guard COMMAND test-runtime-guard)

add_executable(test-audio-lifecycle integration/test_audio_lifecycle.c)
target_include_directories(test-audio-lifecycle PRIVATE support/mocks/audio support/mocks
    "${platform_root}/include" "${platform_root}/src/internal")
target_compile_definitions(test-audio-lifecycle PRIVATE MYBOT_AUDIO_PTIME_MS=20)
target_link_libraries(test-audio-lifecycle PRIVATE mybot::public_headers Threads::Threads)
add_test(NAME audio-lifecycle COMMAND test-audio-lifecycle)
add_executable(test-network-lifecycle integration/test_network_lifecycle.c)
target_include_directories(test-network-lifecycle PRIVATE support/mocks
    "${platform_root}/include" "${platform_root}/src/internal")
target_link_libraries(test-network-lifecycle PRIVATE mybot::public_headers Threads::Threads)
add_test(NAME network-lifecycle COMMAND test-network-lifecycle)

foreach(target test-pixels test-input test-input-worker test-renderer test-identity test-ui-runtime test-drm test-config test-runtime-guard test-audio-lifecycle test-network-lifecycle)
    target_include_directories(${target} PRIVATE "${support_dir}")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
set_tests_properties(pixels input-semantics input-worker renderer identity ui-runtime drm-lifecycle app-config runtime-guard audio-lifecycle network-lifecycle PROPERTIES TIMEOUT 30)
add_test(NAME tooling COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests/tooling -p "test_*.py" -v)
set_tests_properties(tooling PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 120)

    "${CMAKE_CURRENT_SOURCE_DIR}/integration/check_ui_preview.py"
    --binary "$<TARGET_FILE:mybot-ui-preview>" --language "${RV1126B_LANGUAGE}")
set_tests_properties(ui-baseline PROPERTIES TIMEOUT 30)
