target_include_directories(test-input PRIVATE mocks ../platforms/rv1126b ../third_party/mybot/include)
add_test(NAME input-semantics COMMAND test-input)
add_executable(test-input-worker test_input_worker.c ../platforms/rv1126b/input_logic.c)
target_include_directories(test-input-worker PRIVATE mocks ../platforms/rv1126b ../third_party/mybot/include)

    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endforeach()
set_tests_properties(pixels input-semantics input-worker renderer identity ui-runtime PROPERTIES TIMEOUT 30)
set(RV1126B_TEST_DRM_INCLUDE "$ENV{RV1126B_STAGING}/usr/include" CACHE PATH "libdrm test headers")
if(EXISTS "${RV1126B_TEST_DRM_INCLUDE}/xf86drm.h")
    add_executable(test-drm test_drm.c ../ui/pixels.c)
    target_include_directories(test-drm PRIVATE ../ui "${RV1126B_TEST_DRM_INCLUDE}"
        "${RV1126B_TEST_DRM_INCLUDE}/libdrm")
    target_compile_options(test-drm PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME drm-lifecycle COMMAND test-drm)
    set_tests_properties(drm-lifecycle PROPERTIES TIMEOUT 30)
else()
    message(WARNING "DRM mock test omitted: set RV1126B_TEST_DRM_INCLUDE to target libdrm headers")
endif()
